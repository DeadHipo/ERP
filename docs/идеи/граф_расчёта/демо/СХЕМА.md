# Схема данных демо (`cases.js`) и список кейсов

Файл `cases.js` объявляет `window.CASES = [...]`. Его порождает `build-cases.ts`, который прогоняет кейсы через
**настоящие слои** `packages/backend/src/facade/layers` с контекстом, собранным как в `packages/backend/tests/layers/world.ts`
(`makeContext`, `unit`, `allocateUnit`), и выводит рёбра по решениям Р1–Р9 (`../консилиум/итог.md`). Только
синтетические псевдонимы (A, B, R, MC, NG …), никаких реальных имён.

```ts
interface DemoCase {
  id: string;                 // 'G04'
  title: string;              // короткое русское название
  level: 'простой' | 'средний' | 'сложный';
  source: string;             // 'docs/agents/12_golden_cases.md G04'
  description: string;        // 2–4 предложения: что за правила и что ожидаем
  units: DemoUnit[];          // обычно один; G24 — два (трек-месяц из двух юнитов)
  inputs: DemoInput[];        // правила, которыми параметризованы узлы (инспектор Unity)
}
interface DemoUnit {
  unitKey: string;            // `${upc}|${salesMonth}|${labelName}|${account}`
  unit: { isrc; upc; reportingMonth; salesMonth; labelName; account; net: string };
  status: 'allocated' | 'no_split' | 'unknown_pair' | 'no_isrc';
  scheme: 'master/standard';
  nodes: DemoNode[];
  edges: DemoEdge[];
  trace: { layer: string; scaledNet: string; labelShare: string; rows: { artistId; kind; amount: string }[] }[];
  checks: { nodeBalance: boolean; sinksEqualNet: boolean; edgesEqualAllocations: boolean; signOk: boolean };
}
interface DemoNode {
  id: string;                 // 'net' | 'scaledNet' | 'pool' | 'labelShare' | 'ours' | `row:${id}` | `artist:${id}` | `c:${reason}`
  label: string;              // подпись: 'net', 'после top', 'строка A', 'пул', 'доля лейбла', 'A', 'наша доля', 'ждёт: label_unmapped'
  column: 0|1|2|3|4;          // колонка для раскладки: 0 net, 1 scaledNet, 2 row:*/pool, 3 labelShare, 4 стоки
  in: string; out: string;    // decimal-строки, 12 знаков; у стоков out = in
  notes: { ruleId: string; missing: string; why: string }[];   // shortfall подписи
  remainderTo?: string;       // id узла, куда уходит остаток (для подсветки)
}
interface DemoEdge {
  from: string; to: string;
  amount: string;             // decimal-строка 12 знаков, знак юнита или 0
  kind: 'top'|'row'|'draw'|'artist_pool'|'artist_share'|'label_share'|'sublabel'|'remainder'|'perimeter';
  ruleId: string | null; documentId: string | null;
  percent: string | null; base: string | null;     // base = сумма, от которой взят процент
  category: 'B' | 'C' | 'ours' | null;             // только у рёбер в artist:* / ours / c:*
  reason: string | null; assumption: 'document'|'q11'|'r1'|null;
}
interface DemoInput {
  kind: 'unit'|'label'|'split_row'|'participation'|'label_rule'|'document';
  id: string; summary: string;                      // 'участие R 15 % (net, top), документ ok'
  fields: Record<string, string | null>;            // как в базе
  node: string;                                     // узел, который его читает: 'off-the-top', 'split', 'carve:pool' …
}
```

Порядок рёбер: по слою (top, row, draw, artist_pool, artist_share, label_share, sublabel, remainder), затем `to`,
затем `ruleId` (Р8). Нулевые рёбра: есть, если есть аллокация; ребро-остаток есть всегда (Р6).
